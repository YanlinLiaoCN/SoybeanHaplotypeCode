MODEL_NAME = "1B_agro_nt"
# Maximum nucleotide sequence length provided to AgroNT
MAX_SEQ_LEN = 64
# Number of haplotype blocks processed in one CPU batch
BATCH_SIZE = 8
# Fixed GPU inference batch size to improve computational efficiency
FIXED_WINDOW_BATCH = 64

# ============================================================
# 1. Load AgroNT model and compile inference function
# ============================================================
print(f"Initializing AgroNT model: {MODEL_NAME}...")
parameters, forward_fn, tokenizer, config = get_pretrained_model(
    model_name=MODEL_NAME,
    embeddings_layers_to_save=(12,),
    max_positions=MAX_SEQ_LEN
)
forward_fn_transformed = hk.transform(forward_fn)
@jax.jit
def fast_inference(params, key, tokens):
    """
    Perform optimized AgroNT inference.
    The function calculates the sequence-level log likelihood
    from nucleotide language model predictions.
    JIT compilation is applied to accelerate repeated inference
    with fixed input dimensions.
    """
    outputs = forward_fn_transformed.apply(
        params,
        key,
        tokens
    )
    logits = outputs["logits"]
    # Convert logits to nucleotide log probabilities
    log_probs = jax.nn.log_softmax(
        logits,
        axis=-1
    )
    # Extract probabilities of observed nucleotide tokens
    target_tokens = tokens[:, :, None]
    seq_log_probs = jnp.take_along_axis(
        log_probs,
        target_tokens,
        axis=-1
    )
    # Sum nucleotide-level log probabilities
    return jnp.sum(
        seq_log_probs,
        axis=(1, 2)
    ).squeeze()
random_key = jax.random.PRNGKey(42)

# ============================================================
# 2. Build SNP position index
# ============================================================
def build_snp_lookup(vcf_path):
    """
    Create a SNP lookup dictionary from VCF.
    Each SNP is indexed by its identifier and contains:
    chromosome position, reference allele, and alternative allele.
    """
    lookup = {}
    print("Indexing SNP positions from VCF...")
    vcf = pysam.VariantFile(vcf_path)
    for record in vcf:
        # Retain only biallelic SNPs
        if (
            len(record.ref) == 1
            and record.alts
            and len(record.alts[0]) == 1
        ):
            sid = (
                record.id
                if record.id
                else f"{record.chrom}_{record.pos}"
            )
            lookup[sid] = {
                "pos": record.pos - 1,
                "ref": record.ref,
                "alt": record.alts[0]
            }
    vcf.close()
    return lookup

# ============================================================
# 3. Calculate haplotype block-level AgroNT LLR
# ============================================================
def run_optimized_batch(batch_df, fasta, snp_lookup):
    """
    Calculate AgroNT-based haplotype LLR scores.
    For each haplotype block:
    1. Construct the reference haplotype sequence.
    2. Introduce all alternative alleles contained in the block.
    3. Calculate the difference between alternative and reference
       sequence likelihoods.
       Haplotype LLR =
       logP(alternative haplotype)
       -
       logP(reference haplotype)
    """
    all_window_refs = []
    all_window_alts = []
    block_map = []

    # --------------------------------------------------------
    # Extract genomic sequence covering the entire batch region
    # --------------------------------------------------------
    chrom = str(
        batch_df.iloc[0]["chr"]
    )
    batch_min = (
        int(batch_df["Start_pos"].min())
        -
        (MAX_SEQ_LEN // 2)
    )
    batch_max = (
        int(batch_df["End_pos"].max())
        +
        (MAX_SEQ_LEN // 2)
    )
    try:
        full_batch_seq = fasta.fetch(
            chrom,
            batch_min,
            batch_max
        ).upper()
    except Exception as e:
        print(
            f"Sequence retrieval failed for "
            f"{chrom}:{batch_min}-{batch_max}: {e}"
        )
        return None

    # --------------------------------------------------------
    # Generate reference and alternative haplotype sequences
    # --------------------------------------------------------
    for idx, row in batch_df.iterrows():
        snp_ids = [
            s.strip()
            for s in str(row["SNPs_in_block"]).split(",")
        ]
        current_block_snps = []
        for sid in snp_ids:
            if sid in snp_lookup:
                current_block_snps.append(
                    snp_lookup[sid]
                )
        if not current_block_snps:
            continue
        # Sort SNPs according to genomic position
        current_block_snps.sort(
            key=lambda x: x["pos"]
        )
        snp_positions = [
            s["pos"]
            for s in current_block_snps
        ]
        # Define non-overlapping sequence windows
        block_windows = []
        last_w_end = -1
        for pos in snp_positions:
            if pos < last_w_end:
                continue
            w_start = max(
                0,
                pos - (MAX_SEQ_LEN // 2)
            )
            w_end = w_start + MAX_SEQ_LEN
            block_windows.append(
                (w_start, w_end)
            )
            last_w_end = w_end
        current_start_idx = len(all_window_refs)
        for w_start, w_end in block_windows:
            rel_start = w_start - batch_min
            rel_end = w_end - batch_min
            ref_seq = full_batch_seq[
                rel_start:rel_end
            ]
            # Padding sequence to fixed length
            if len(ref_seq) < MAX_SEQ_LEN:
                ref_seq = ref_seq.ljust(
                    MAX_SEQ_LEN,
                    "N"
                )
            # Introduce all alternative alleles
            alt_seq_chars = list(ref_seq)
            has_applied = False
            for snp in current_block_snps:
                rel_pos = snp["pos"] - w_start
                if 0 <= rel_pos < MAX_SEQ_LEN:
                    if ref_seq[rel_pos] == snp["ref"]:
                        alt_seq_chars[rel_pos] = (
                            snp["alt"]
                        )
                        has_applied = True
            if has_applied:
                all_window_refs.append(
                    ref_seq
                )
                all_window_alts.append(
                    "".join(alt_seq_chars)
                )
        block_map.append(
            (
                idx,
                current_start_idx,
                len(all_window_refs)
            )
        )
    if not all_window_refs:
        return None

    # --------------------------------------------------------
    # Tokenization and model inference
    # --------------------------------------------------------
    combined_seqs = (
        all_window_refs
        +
        all_window_alts
    )
    tokens_out = tokenizer.batch_tokenize(
        combined_seqs
    )
    raw_token_ids = np.zeros(
        (
            len(combined_seqs),
            MAX_SEQ_LEN
        ),
        dtype=np.int32
    )
    for i, t in enumerate(tokens_out):
        ids = t[1][:MAX_SEQ_LEN]
        raw_token_ids[
            i,
            :len(ids)
        ] = ids
    jax_token_ids = jnp.array(
        raw_token_ids
    )
    # GPU inference using fixed-size batches
    all_scores_list = []
    num_total_windows = len(combined_seqs)
    for i in range(
        0,
        num_total_windows,
        FIXED_WINDOW_BATCH
    ):
        chunk = jax_token_ids[
            i:i + FIXED_WINDOW_BATCH
        ]
        actual_len = len(chunk)
        # Padding the last batch
        if actual_len < FIXED_WINDOW_BATCH:
            padding = jnp.zeros(
                (
                    FIXED_WINDOW_BATCH - actual_len,
                    MAX_SEQ_LEN
                ),
                dtype=jnp.int32
            )
            chunk = jnp.concatenate(
                [
                    chunk,
                    padding
                ],
                axis=0
            )
        chunk_scores = fast_inference(
            parameters,
            random_key,
            chunk
        )
        all_scores_list.append(
            np.array(chunk_scores)[:actual_len]
        )
    all_scores = np.concatenate(
        all_scores_list
    )

    # --------------------------------------------------------
    # Calculate haplotype LLR
    # --------------------------------------------------------
    num_refs = len(all_window_refs)
    llrs = (
        all_scores[num_refs:]
        -
        all_scores[:num_refs]
    )
    batch_results = []
    for orig_idx, start, end in block_map:
        block_window_llrs = llrs[start:end]
        if len(block_window_llrs) > 0:
            result = batch_df.loc[
                orig_idx
            ].to_dict()
            result["Haplotype_LLR"] = float(
                np.mean(block_window_llrs)
            )
            result["Num_Windows"] = int(
                end - start
            )
            batch_results.append(
                result
            )
    return batch_results
